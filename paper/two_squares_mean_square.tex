\title[Sums of two squares with few prime factors, II]{Sums of two squares with few prime factors, II:\\ the mean square of the cuspidal error of $\theta^2(mz)\theta^2(m'z)$,\\ almost all $N$, and the size of the Petersson norm}

\begin{abstract}
Let $m,m'$ be coprime odd integers, $\Theta_{m,m'}(z)=\theta^2(mz)\theta^2(m'z)$, $G_{m,m'}$ its cuspidal part and $L=4mm'$, so that $a_{G}(N)=T(m,m';N)-M(m,m';N)$ is the error term in the number of representations of $N$ by $m(x^2+y^2)+m'(z^2+w^2)$. Part~I proved $\|G_{m,m'}\|^2\ll_\e(mm')^{\e}$ for the Petersson norm and deduced the level of distribution $1/2$ for the weighted sequence $r(n)r(N-n)$ and the bounds $\Omega(n(N-n))\le8$, $\max(\Omega(n),\Omega(N-n))\le5$ for every large $N\equiv2\pmod4$. Here we prove a mean-square bound $\sum_{N\le Y}|a_G(N)|^2\le4\pi e^2Y(1+54Y/L)\|G\|^2$ for every $G\in S_2(\Gamma_0(L))$ by a covering argument; for $G_{m,m'}$ it says that the root mean square of $(T-M)/\sqrt N$ is $\ll(mm')^{-1/2+\e}$, and it gives the level of distribution $1-\e$ for the weighted sequence for all but $O(X^{1-\delta})$ integers $N\le X$, and for those $N$ the bounds $5$ and $4$ (a result which, as an arithmetic statement about almost all $N$, is superseded by the circle method, which gives $N=p+q$ with $p\equiv q\equiv1\pmod 4$). We reformulate the uniform bound of Part~I in terms of Poincar\'e series, state the individual bound $|T-M|\ll N^{1/2+\e}(mm')^{-1/2+\e}$ suggested by the data, and show that any power saving in $mm'$ for individual $N$ raises the level of distribution for every $N$ beyond $1/2$. Numerically, $\|G_{m,m'}\|^2\approx0.37\,(\log mm')^2$, with an exact check at the level $20$, and the supremum of $|T-M|/\sqrt N$ over $N$ exceeds the root mean square by a bounded factor at every level computed.
\end{abstract}

\section{Results}\label{sec:results}

We keep the notation of \cite{ChenI}, referred to as Part~I. Thus $r(n)$ is the number of representations of $n$ as a sum of two squares, $m,m'$ are coprime odd positive integers,
\begin{gather*}
T(m,m';N)=\sum_{ma+m'b=N}r(a)r(b),\qquad L=4mm',\\
\Theta_{m,m'}(z)=\theta^2(mz)\theta^2(m'z)=\sum_{N\ge0}T(m,m';N)e(Nz)\in M_2(\Gamma_0(L)),
\end{gather*}
$\Theta_{m,m'}=E+G$ is the decomposition into the Eisenstein part, whose coefficients are the Siegel main terms $M(m,m';N)$, and the cuspidal part $G=G_{m,m'}\in S_2(\Gamma_0(L))$, so that
\[
a_G(N)=T(m,m';N)-M(m,m';N)\qquad\text{for every }N\ge1 .
\]
The Petersson norm is $\|G\|^2=\int_{\Gamma_0(L)\backslash\HH}|G|^2\,\dd x\,\dd y$. Lemma~2.1 of Part~I, which is Waibel's theorem \cite[Theorem~1]{Waibel} for the form $Q_{m,m'}=m(x^2+y^2)+m'(z^2+w^2)$, gives $\|G\|\ll(mm')^{1/4+\e}$, and the first two versions of Part~I deduced from it, through a Hecke operator, the uniform bound $a_G(N)\ll d(N)N^{1/2}(mm')^{1/4+\e}$ for $(N,mm')=1$, $v_2(N)\le1$, the level of distribution $2/5$ and the bounds $9$ and $6$, and conjectured (Conjecture~5.1 there) that the factor $(mm')^{1/4}$ can be removed. Version~3 of Part~I, which this paper accompanies, proves that the norm itself is much smaller than Waibel's general bound, by computing the expansion of $\Theta_{m,m'}$ at every cusp of $\Gamma_0(L)$ with Waibel's method, and draws the consequences. We recall what we use.

\begin{theorem}[Part~I, Theorems~1.2 and 2.3]\label{thm:norm}
Let $m,m'$ be coprime odd positive integers.
\begin{enumerate}[label=\textup{(\roman*)},leftmargin=2.2em]
\item $\|G_{m,m'}\|^2\ll_\e(mm')^{\e}$.
\item For $N\ge1$ with $(N,mm')=1$ and $v_2(N)\le1$,
\[
T(m,m';N)=M(m,m';N)+O_\e\bigl(d(N)\,N^{1/2}(mm')^{\e}\bigr);
\]
for cube-free $m,m'$ and every $N$ with $v_2(N)\le1$ the error term is $O_\e(d(N)^4N^{1/2}(mm')^\e)$ (Part~I, Corollary~2.7).
\end{enumerate}
\end{theorem}

Part~I deduces from (ii) that $r(n)r(N-n)$ has level of distribution $1/2$ (Theorem~3.2 there) and, with sieve parameters certified in interval arithmetic, that for every sufficiently large $N\equiv2\pmod4$
\begin{multline*}
\#\bigl\{1\le n<N:\ n=x^2+y^2,\ N-n=z^2+w^2,\ \Omega(n(N-n))\le 8,\ \max(\Omega(n),\Omega(N-n))\le 5\bigr\}\\
\ge\ 0.23\,\Sg_r(N)\,\frac{N}{(\log N)^2},\qquad \Sg_r(N)=\prod_{\substack{p\mid N\\ p>2}}\frac{p-1}{p-2},
\end{multline*}
(Theorem~1.1 there; versions 1--2 had $9$ and $6$). The proof of (i) shows that the expansion of $\Theta_{m,m'}$ at every cusp is again a theta series of a form $A(x_1^2+x_2^2)+B(x_3^2+x_4^2)$ with $AB\asymp mm'$, whose coefficients are $O(n^\e)$ for $n\ll mm'$ because $A\alpha+B\beta=n$ has $O(1)$ solutions, and suggests $\|G_{m,m'}\|^2\asymp(\log mm')^2$ (Part~I, Remark~2.13); numerically, $\|G_{m,m'}\|^2\approx0.37\,(\log mm')^2$ (Section~\ref{sec:num}).

The first theme of this paper is the average over $N$ of the error term, which is of independent interest and gives a stronger level of distribution for almost all $N$.

\begin{theorem}[Mean square]\label{thm:A}
Let $L\ge1$, $G\in S_2(\Gamma_0(L))$ and $Y\ge1$. Then
\begin{equation}\label{eq:A}
\sum_{n\le Y}|a_G(n)|^2\ \le\ 4\pi e^2\,Y\Bigl(1+\frac{54\,Y}{L}\Bigr)\|G\|^2 .
\end{equation}
In particular, for coprime odd $m,m'$, $G=G_{m,m'}$ and $Y\ge mm'$, by Theorem~\ref{thm:norm}(i),
\begin{equation}\label{eq:A2}
\sum_{N\le Y}\bigl|T(m,m';N)-M(m,m';N)\bigr|^2\ \ll_\e\ Y^2\,(mm')^{-1+\e},
\end{equation}
and hence, since $\sum_{Y/2<N\le Y}|a_G(N)|^2/N\le(2/Y)\sum_{N\le Y}|a_G(N)|^2$, the root mean square of $(T-M)/\sqrt N$ over $Y/2<N\le Y$ is $\ll(mm')^{-1/2+\e}$.
\end{theorem}

The proof of \eqref{eq:A} is a classical covering argument and does not use the Petersson formula; the dependence on the level is exactly the factor $Y/L$, i.e.\ the volume of $\Gamma_0(L)\backslash\HH$. The bound \eqref{eq:A2} is the exact order of magnitude: by the Rankin--Selberg method the mean square of $(T-M)/\sqrt N$ is asymptotic to $16\pi^2\|G\|^2/\vol(\Gamma_0(L)\backslash\HH)$, and Table~\ref{tab:ms} shows the data.

Averaging over $N$ is natural for the arithmetic application, since the sieve of Part~I is applied to one $N$ at a time and an exceptional set of $N$ can be tolerated. Let, as in Part~I, $N\equiv2\pmod4$, $\cA=\{n\le N:\ (n,N)=1\}$, $w(n)=r(n)r(N-n)$, and for coprime square-free $d,e$ with $(de,2N)=1$ let $R(d,e)=R(d,e;N)$ be the remainder in
\[
\sum_{\substack{n\in\cA\\ d\mid n,\ e\mid N-n}}w(n)=X_N\,h(d)h(e)+R(d,e;N)
\]
(this is $S(d,e)=M(d,e)+R(d,e)$ of Part~I, Section~3; we write $X_N$ for the main term $X$ there, to avoid a clash with the length $X$ of the range of $N$ below).

\begin{theorem}[Level of distribution $1-\e$ for almost all $N$]\label{thm:B}
Let $0<\e<1/20$, $X\ge X_0(\e)$ and $D=X^{1-3\e}$. Then
\[
\sum_{\substack{X/2<N\le X\\ N\equiv2\ (4)}}\ \sum_{\substack{de\le D,\ (d,e)=1\\ (de,2N)=1}}\mu^2(de)\,d_3(de)\,|R(d,e;N)|\ \ll_\e\ X^{2-\e/2}.
\]
Consequently, for all but $O_\e(X^{1-\e/4})$ integers $N\equiv2\pmod4$ in $(X/2,X]$,
\begin{equation}\label{eq:B}
\sum_{\substack{de\le N^{1-3\e}\\ (d,e)=1,\ (de,2N)=1}}\mu^2(de)\,d_3(de)\,|R(d,e;N)|\ \le\ N^{1-\e/5}.
\end{equation}
\end{theorem}

The sieve of Part~I, Section~4, uses the level of distribution only through \eqref{eq:B}; running it at the level $\theta=19/20$ gives the following.

\begin{corollary}\label{cor:C}
For all but $O(X^{1-\delta})$ integers $N\equiv2\pmod 4$ in $(X/2,X]$, with an absolute $\delta>0$,
\begin{multline*}
\#\bigl\{1\le n<N:\ n=x^2+y^2,\ N-n=z^2+w^2,\ \Omega(n(N-n))\le 5,\ \max(\Omega(n),\Omega(N-n))\le 4\bigr\}\\
\ge\ 0.34\,\Sg_r(N)\,\frac{N}{(\log N)^2},
\end{multline*}
the integers counted being coprime to $N$ and products of distinct primes $\equiv1\pmod4$ exceeding $N^{2/25}$.
\end{corollary}

The certified sieve parameters are $\alpha=2/25$, $\lambda=1/3$, $\beta_1=69/100$ (Section~\ref{sec:B}). The pairs $(5,4)$ for almost all $N$ and $(8,5)$ for every $N$ should be compared with $(9,6)$ for every $N$, which is what Waibel's bound alone gives (Part~I, Remark~4.5). A floating-point scan (Figure~\ref{fig:thr}) shows that the best attainable value of $1/\lambda+1/\beta_1$ drops below $4$ only for $\theta>0.88$; at $\theta=19/20$ the scan gives $3.83$ (at $\alpha=0.10$, $\beta_1=0.47$, $\lambda=0.59$, with $1/\lambda+2/\beta_1=5.96$), i.e.\ the pair $(5,3)$, which we have not certified, for the reason explained next.

Corollary~\ref{cor:C} must also be compared with what the circle method gives for almost all $N$. By the method of Montgomery and Vaughan \cite{MV}, carried out with the characters modulo $4$ (see \cite{BauerWang} for Goldbach's problem with primes in prescribed residue classes), all but $O(X^{1-\delta})$ integers $N\equiv2\pmod4$ up to $X$ are sums $p+q$ of two primes $p\equiv q\equiv1\pmod 4$, with $\gg\Sg_r(N)N/(\log N)^2$ representations; such primes are sums of two squares, so for almost all $N$ the pair $(1,1)$ holds with the count of Corollary~\ref{cor:C}. As an arithmetic statement about almost all $N$, Corollary~\ref{cor:C} is therefore far from new. What is new is Theorem~\ref{thm:B}, a level of distribution $1-\e$ for the weighted sequence $r(n)r(N-n)$, together with the observation that the sieve of Part~I extracts $(5,4)$ from it with no further input; the sieve route adds something only for \emph{every} $N$, where the circle method gives nothing (Goldbach's problem) and where Part~I gives $(8,5)$. Section~\ref{sec:spectral} formulates the individual bound that would raise the level to $1-\e$ for every $N$ and with it give $(5,4)$ for every $N$ (Proposition~\ref{prop:beyond}).

Section~\ref{sec:spectral} restates the uniform bound of Theorem~\ref{thm:norm}(ii) in terms of Poincar\'e series and formulates the individual bound, and Section~\ref{sec:num} contains the numerical evidence: the data of Part~I, an exact check of the whole numerical chain at the level $20$, the empirical law $\|G_{m,m'}\|^2\approx0.37(\log mm')^2$, which corrects a remark in the first two versions of Part~I, and the supremum of $|T-M|/\sqrt N$ over $N$.

\section{Proof of Theorem~\ref{thm:A}}\label{sec:A}

Let $G(z)=\sum_{n\ge1}a(n)e(nz)\in S_2(\Gamma_0(L))$ and put $F(z)=y^2|G(z)|^2$, $z=x+iy$. Since $G(\gamma z)=(cz+d)^2G(z)$ and $\operatorname{Im}(\gamma z)=y/|cz+d|^2$ for $\gamma=\left(\begin{smallmatrix}a&b\\ c&d\end{smallmatrix}\right)\in\Gamma_0(L)$, the function $F$ is $\Gamma_0(L)$-invariant, and with the invariant measure $\dd\mu=y^{-2}\dd x\,\dd y$ we have $\|G\|^2=\int_{\cF}F\,\dd\mu$ for any fundamental domain $\cF$ of $\Gamma_0(L)$.

Let $y_0=1/(4\pi Y)$ and $R=\{x+iy:\ 0\le x<1,\ y_0\le y\le2y_0\}$ (half-open in $x$, so that each orbit of the translations $z\mapsto z+b$ meets $R$ at most once). By Parseval, $\int_0^1|G(x+iy)|^2\,\dd x=\sum_n|a(n)|^2e^{-4\pi ny}$, hence
\[
\int_R|G|^2\,\dd x\,\dd y=\sum_n|a(n)|^2\int_{y_0}^{2y_0}e^{-4\pi ny}\,\dd y\ \ge\ \frac{e^{-2}}{4\pi Y}\sum_{n\le Y}|a(n)|^2 ,
\]
because $4\pi ny\le 8\pi Yy_0=2$ for $n\le Y$, $y\le2y_0$. On the other hand $\int_R|G|^2\,\dd x\,\dd y=\int_RF\,\dd\mu$, and
\[
\int_RF\,\dd\mu=\sum_{\gamma\in\Gamma_0(L)/\{\pm1\}}\int_{R\cap\gamma\cF}F\,\dd\mu=\int_{\cF}F(w)\,\#\{\gamma:\ \gamma w\in R\}\,\dd\mu(w)\ \le\ \mu_R\,\|G\|^2,
\]
where $\mu_R=\sup_{z\in R}\#\{\gamma\in\Gamma_0(L)/\{\pm1\}:\ \gamma z\in R\}$ (if $\#\{\gamma:\gamma w\in R\}\ne0$ for some $w\in\cF$, pick $\gamma_0$ with $z=\gamma_0w\in R$; then $\gamma\mapsto\gamma\gamma_0^{-1}$ is a bijection from $\{\gamma:\gamma w\in R\}$ onto $\{\gamma:\gamma z\in R\}$, so the count at $w$ is at most $\mu_R$). It remains to bound $\mu_R$. Fix $z=x+iy\in R$ and let $\gamma=\left(\begin{smallmatrix}a&b\\ c&d\end{smallmatrix}\right)\in\Gamma_0(L)$ with $\gamma z\in R$.
\begin{itemize}[leftmargin=2em]
\item If $c=0$, then $\gamma=\pm\left(\begin{smallmatrix}1&b\\ 0&1\end{smallmatrix}\right)$ and $\gamma z=z+b$, so $x+b\in[0,1)$ forces $b=0$: one element.
\item If $c\ne0$, we may take $c>0$, since $\gamma$ and $-\gamma$ are the same element of $\Gamma_0(L)/\{\pm1\}$. Then $L\mid c$, and $\operatorname{Im}(\gamma z)=y/|cz+d|^2\in[y_0,2y_0]$ together with $y\in[y_0,2y_0]$ gives $|cz+d|^2\in[1/2,2]$. Since $|cz+d|^2=(cx+d)^2+c^2y^2$, this forces $c^2y^2\le2$, hence $c\le\sqrt2/y_0=4\sqrt2\,\pi Y$, and $|cx+d|\le\sqrt2$. The number of positive multiples $c$ of $L$ with $c\le4\sqrt2\pi Y$ is $\lfloor4\sqrt2\pi Y/L\rfloor\le4\sqrt2\pi\,Y/L$; for each $c$ there are at most $3$ integers $d$ with $|cx+d|\le\sqrt2$; and for each pair $(c,d)$ the elements of $\Gamma_0(L)$ with bottom row $(c,d)$ are $\left(\begin{smallmatrix}1&k\\ 0&1\end{smallmatrix}\right)\gamma_0$, $k\in\Z$, whose images $\gamma_0z+k$ have real part in $[0,1)$ for exactly one $k$. This gives at most $4\sqrt2\pi\cdot3\,Y/L=12\sqrt2\pi\,Y/L<54\,Y/L$ elements.
\end{itemize}
Hence $\mu_R\le1+54Y/L$, and
\[
\sum_{n\le Y}|a(n)|^2\le 4\pi e^2Y\int_RF\,\dd\mu\le 4\pi e^2\,Y\Bigl(1+\frac{54Y}L\Bigr)\|G\|^2,
\]
which is \eqref{eq:A}. For $G=G_{m,m'}$, $L=4mm'$ and $Y\ge mm'$ the bracket is $\le 1+54Y/(4mm')\le 15Y/(mm')$, and Theorem~\ref{thm:norm}(i) gives \eqref{eq:A2}. \qed

\begin{remark}\label{rem:rankin}
The argument is the classical one for bounding $\sum_{n\le Y}|a(n)|^2$ by the norm (compare \cite[Chapter~5]{IwaTopics}); its only input is the invariance of $y^2|G|^2$. The strip $y_0\le y\le2y_0$ with $y_0=1/(4\pi Y)$ is chosen so that the exponential factor in Parseval's identity stays at $e^{-2}$ for all $n\le Y$. For $Y\ge L$ the factor $Y^2/L$ in \eqref{eq:A} is of the correct order of magnitude: by the Rankin--Selberg method, $\sum_{n\le Y}|a(n)|^2/n\sim 16\pi^2\|G\|^2Y/\vol(\Gamma_0(L)\backslash\HH)$ with $\vol=\frac\pi3L\prod_{p\mid L}(1+1/p)$ (measure $\dd\mu$), so that $\sum_{n\le Y}|a(n)|^2\sim 8\pi^2\|G\|^2Y^2/\vol$ for fixed $G$ as $Y\to\infty$. Theorem~\ref{thm:A} makes this uniform in the level, at the cost of the constant.
\end{remark}

\section{Proof of Theorem~\ref{thm:B} and Corollary~\ref{cor:C}}\label{sec:B}

\subsection{The remainder on average}
We follow the proof of Theorem~3.2 of Part~I, replacing the uniform bound for $|T-M|$ by Theorem~\ref{thm:A} after an application of Cauchy--Schwarz over $N$. Recall from Part~I, (15), that for $N\equiv2\pmod4$ and coprime square-free $d,e$ with $(de,2N)=1$,
\begin{equation}\label{eq:Rdecomp}
R(d,e;N)=\sum_{t\mid N^\flat}\mu(t)\sum_{m\mid d^2}\sum_{m'\mid e^2}c_d(m)c_e(m')\sum_{\substack{t_1\mid t^2\\ t\mid t_1}}c_t(t_1)\,\bigl(a_{G}(N)-a_{G}(N/2)\bigr),\qquad G=G_{t_1m,m'},
\end{equation}
where $N^\flat$ is the product of the odd primes dividing $N$, $m=ds$, $m'=es'$ with $s\mid d$, $s'\mid e$, and likewise $t_1=tu$ with $u\mid t$ (the coefficients $c_d(m)$, $c_t(t_1)$ of Part~I, (14), vanish unless $m=ds$ with $s\mid d$, resp.\ $t_1=tu$ with $u\mid t$; in particular $t\mid t_1$); the parameters $t_1m,m'$ of the form $Q_{t_1m,m'}$ are coprime, odd and cube-free, $\tilde m\tilde m':=t_1mm'=t_1ss'\cdot de$, and the coefficients $c$ satisfy $|c_d(m)c_e(m')c_t(t_1)|\le3^{\omega(d)+\omega(e)+\omega(t)}$. (For the reader's convenience: $c_d$ is the multiplicative coefficient of Part~I, (14), with $c_p(p)=2$, $c_p(p^2)=-1$ for a split prime $p$ and $c_q(q)=0$, $c_q(q^2)=1$ for an inert prime $q$; it encodes the identities $r(n)=2r(n/p)-r(n/p^2)$ and $r(n)=r(n/q^2)$, which turn the condition $d\mid n$ into a combination of theta series $\theta^2(mz)$. The sum over $t$ is the M\"obius inversion of the condition $(n,N)=1$, and $a_G(N)-a_G(N/2)$ selects the odd $n$.) Let $\eta=\e/10$. For $N\le X$ with $X\ge X_0(\e)$ we have, as in (F1)--(F3) of Part~I: $d_3(de)\,3^{\omega(d)+\omega(e)+\omega(t)}\,4^{\omega(N)}d(N)^2\le X^\eta$; for fixed $(d,e,N)$ the number of tuples $(s,s',t,t_1)$ is at most $X^\eta$; and $\sum_{k\le D}d(k)/k\le X^\eta$, $\sum_{k\le D}d(k)k^{-1/2}\le D^{1/2}X^{\eta}$. We also use the two bounds for a single term,
\begin{gather}
|a_G(N')|\le T(\tilde m,\tilde m';N')+M(\tilde m,\tilde m';N')\ll X^{2\eta}\Bigl(\frac{X}{\tilde m\tilde m'}+1\Bigr)\qquad(N'\le X),\label{eq:triv}\\
\sum_{n\le X}|a_G(n)|\le X^{1/2}\Bigl(\sum_{n\le X}|a_G(n)|^2\Bigr)^{1/2}\ll X\Bigl(1+\frac XL\Bigr)^{1/2}L^{\eta},\label{eq:ms}
\end{gather}
where $G=G_{\tilde m,\tilde m'}$ and $L=4\tilde m\tilde m'$, the first being (20) of Part~I, the second Cauchy--Schwarz, \eqref{eq:A} and Theorem~\ref{thm:norm}(i); for $L\le X$ the right-hand side of \eqref{eq:ms} is $\ll X^{3/2}(\tilde m\tilde m')^{-1/2+\eta}$. Let $\Sigma$ denote the double sum in Theorem~\ref{thm:B}. Inserting \eqref{eq:Rdecomp} and splitting the tuples according to the size of $t_1ss'$:

\emph{Tuples with $t_1ss'>X^\e$.} For each $N$ and each tuple, \eqref{eq:triv} with $\tilde m\tilde m'>X^\e de$ gives $|a_G(N')|\ll X^{2\eta}(X^{1-\e}/(de)+1)$. Summing over the $\le X^\eta$ tuples, over $N\le X$ and over $de\le D$ with the weights $d_3(de)\cdot3^{\omega}\le X^\eta$, these tuples contribute
\[
\ll X^{1+4\eta}\sum_{k\le D}d(k)\Bigl(\frac{X^{1-\e}}{k}+1\Bigr)\ll X^{2-\e+5\eta}+X^{1+5\eta}D\ll X^{2-\e/2}.
\]

\emph{Tuples with $t_1ss'\le X^\e$.} Now $t\le t_1\le X^\e$ (as $t\mid t_1$), so $t$ runs over at most $X^\e$ values, $t_1$ over at most $d(t^2)\le X^\eta$ values, and $(s,s')$ over at most $d(d)d(e)\le X^\eta$ values; the forms $G=G_{t_1m,m'}$ have level $L=4t_1ss'de\le4X^\e D\le X$, so that the right-hand side of \eqref{eq:ms} is $\ll X^{3/2}(\tilde m\tilde m')^{-1/2+\eta}$. For a fixed tuple and fixed $(d,e)$, we drop the condition $t\mid N$ except for the count: the number of $N\le X$ with $t\mid N$ is $\le X/t$, and Cauchy--Schwarz gives
\begin{align*}
\sum_{\substack{N\le X\\ t\mid N}}\bigl(|a_G(N)|+|a_G(N/2)|\bigr)&\le2\Bigl(\frac Xt\Bigr)^{1/2}\Bigl(\sum_{n\le X}|a_G(n)|^2\Bigr)^{1/2}\\
&\ll\Bigl(\frac Xt\Bigr)^{1/2}X\,(\tilde m\tilde m')^{-1/2+\eta}\le X^{3/2+\eta}\,t^{-1/2}(de)^{-1/2},
\end{align*}
using \eqref{eq:A} in the form $\sum_{n\le X}|a_G(n)|^2\ll X^2\|G\|^2/L$ and $(\tilde m\tilde m')^{-1/2+\eta}\le X^{\eta}(de)^{-1/2}$. Summing over $t\le X^\e$ ($\sum_t t^{-1/2}\le2X^{\e/2}$), over the other tuple variables ($\le X^{2\eta}$), and over $de\le D$ with the weights $\le X^\eta$,
\[
\ll X^{3/2+4\eta+\e/2}\sum_{k\le D}d(k)k^{-1/2}\ll X^{3/2+5\eta+\e/2}D^{1/2}=X^{2-\e+5\eta}=X^{2-\e/2},
\]
since $D^{1/2}=X^{1/2-3\e/2}$. Both contributions are $\ll X^{2-\e/2}$, which proves the first assertion. (The margin is small but positive: the choice $D=X^{1-3\e}$ saves $X^{-3\e/2}$ in $D^{1/2}$, of which the sum over $t$ costs $X^{\e/2}$ and the divisor-type factors cost $X^{5\eta}=X^{\e/2}$; this is why $\eta=\e/10$.)

For the second, let $S(N)$ be the inner sum with $D=X^{1-3\e}$. The number of $N\in(X/2,X]$, $N\equiv2\pmod 4$, with $S(N)>X^{1-\e/4}$ is at most $\Sigma/X^{1-\e/4}\ll X^{1-\e/4}$. For the remaining $N$, since $N^{1-3\e}\le D$ and $X^{1-\e/4}\le(2N)^{1-\e/4}\le N^{1-\e/5}$ for large $X$, \eqref{eq:B} holds. \qed

\begin{remark}
The exponent $1$ is the balance $X^{3/2}D^{1/2}=X^{2}$ between the mean-square bound $X^{3/2}(de)^{-1/2}$ for the sum over $N$ of one remainder and the number $D$ of moduli; with a norm bound $\|G_{m,m'}\|\ll(mm')^{\beta+\e}$ in place of Theorem~\ref{thm:norm}(i) one gets the level $1/(2(1+\beta))$ for every $N$ (Part~I, Remark~3.3) and $1/(1+2\beta)$ for almost all $N$; Waibel's $\beta=1/4$ gives $2/5$ and $2/3$, and $\beta=0$ gives $1/2$ and $1$. The exceptional set $X^{1-\e/4}$ comes from Markov's inequality applied to a first moment of $S(N)$, which in turn rests on the second moment of $a_G(N)$; a bound for a higher moment would shrink it, and we do not pursue this.
\end{remark}

\subsection{The sieve at the level $19/20$}
Proposition~4.3 of Part~I and its proof use the level of distribution only through the bound \eqref{eq:B} (there with $2/5$); they hold verbatim, for a given $N$, with $\theta=2/5$ replaced by any $\theta\in(0,1)$ for which \eqref{eq:B} holds with $N^{\theta-\e}$ in place of $N^{1-3\e}$, provided $\alpha<\beta_1<\theta$. Lemmas~4.1 and 4.2 of Part~I do not involve the level at all. Hence, for every $N$ satisfying \eqref{eq:B} with $3\e<1/20$, the conclusion of Proposition~4.3 holds with $\theta=19/20$, and the proof of Theorem~1.1 of Part~I goes through with the parameters certified for $\theta=19/20$. The script \texttt{certify\_modular.py} of Part~I, extended by the argument \texttt{19/20} in the repository of this paper, certifies in interval arithmetic
\begin{gather*}
\theta=\tfrac{19}{20},\qquad\alpha=\tfrac2{25},\qquad\lambda=\tfrac13,\qquad\beta_1=\tfrac{69}{100},\\
\tfrac1\lambda+\tfrac2{\beta_1}=\tfrac{407}{69}<6,\qquad \tfrac1\lambda+\tfrac1{\beta_1}=\tfrac{307}{69}<5,\qquad L-\lambda R\ge0.085058
\end{gather*}
(with $L\ge0.999484$, $R\le2.743278$, $\theta_A=0.4740$; the output files are listed at the end of the paper). By Lemma~4.2 of Part~I this gives $\Omega(n(N-n))\le5$ and $\max(\Omega(n),\Omega(N-n))\le4$, and the weight of each $n$ counted is $w(n)=16\cdot2^{\Omega(n(N-n))}\le2^{9}$. As in the proof of Theorem~1.1 of Part~I, the number of integers counted is at least
\[
\bigl(0.085058+o(1)\bigr)\,2^{-9}\cdot8\,\Sg_r(N)\,c_{2h}\,\frac{N}{\alpha^2(\log N)^2}\ \ge\ 0.34\,\Sg_r(N)\frac{N}{(\log N)^2},
\]
since $0.085058\cdot2^{-9}\cdot8\cdot1.66485\cdot(25/2)^2=0.3457\ldots$. The exceptional set is that of Theorem~\ref{thm:B} with, say, $\e=1/80$, so $\delta=1/320$ is admissible. \qed

\begin{remark}
The same sieve gives $(8,5)$ at the level $1/2$ (Part~I), $(6,4)$ at $2/3$ (certified: $\alpha=3/50$, $\lambda=1/3$, $\beta_1=51/100$, $L-\lambda R\ge0.036741$) and $(5,4)$ at $19/20$. The floating-point scan of Figure~\ref{fig:thr} shows that $1/\lambda+2/\beta_1$ drops below $6$ only for $\theta>0.83$, and that the best attainable value of $1/\lambda+1/\beta_1$ drops below $4$ for $\theta>0.88$: at $\theta=19/20$ the minimum is $3.83$, attained at $\alpha=0.10$, $\beta_1=0.47$, $\lambda=0.59$, where $1/\lambda+2/\beta_1=5.96$, so that the Richert weights give $(5,3)$ for almost all $N$ in the floating-point model, with the sifting parameter $\alpha$ at the edge of the scanned range. We have not certified this set, because Corollary~\ref{cor:C} is in any case superseded for almost all $N$ by the circle method (Section~\ref{sec:results}); it is an ``almost all $N$'' statement with a power-saving exceptional set and says nothing about any individual $N$, which is where the sieve route matters (Section~\ref{subsec:ind}).
\end{remark}

\begin{remark}[The vector sieve is not the natural sieve here]\label{rem:dim2}
The sieve of Part~I treats the two sides $n$ and $N-n$ by the vector sieve of Br\"udern and Fouvry, with two linear components of levels $N^{\theta_A}$ and $N^{\theta-\theta_A}$ and the lower bound $L=f_1(s_A)F_1(s_B)+F_1(s_A)f_1(s_B)-F_1(s_A)F_1(s_B)$. For $s_A,s_B\le3$ one has $f_1/F_1(s)=\log(s-1)$, so $L>0$ if and only if $(s_A-1)(s_B-1)>e$, and with $s=s_A+s_B$ fixed this requires $s>2+2\sqrt e=5.297\ldots$, attained at the symmetric split. The product sequence $n(N-n)$, however, is a sieve problem of dimension $2$ in the classical sense: under the weight $r(n)r(N-n)$ the density of $p\mid n(N-n)$ is $2h(p)$, $\sum_{p<z}2h(p)=2\log\log z+O(1)$, the condition $d\mid n(N-n)$ with $(n,N)=1$ splits uniquely as $d=d_1d_2$, $d_1\mid n$, $d_2\mid N-n$, and the remainder for the modulus $d$ is $\sum_{d_1d_2=d}R(d_1,d_2)$, which is exactly what Theorem~3.2 of Part~I and Theorem~\ref{thm:B} bound (this is the reason for the weight $d_3(de)$ there). The two-dimensional sieve of Diamond, Halberstam and Richert \cite{DH} applies to it directly, with the joint level $d\le N^{\theta}$ and the sieving limit $\beta_2=4.2665\ldots$ in place of $5.297$; its upper and lower functions $F_2,f_2$ should be used in place of the products of $F_1,f_1$ both in $L$ and in the weight term $R$. We expect this to improve the pairs $(k,k')$ at every level without any new analytic input, and we have not carried it out; the bound for $\max(\Omega(n),\Omega(N-n))$ then needs a separate argument, since the two-dimensional sieve controls only $\Omega(n(N-n))$. For comparison, for the unweighted sequence $n(N-n)$, which has level of distribution $1$ trivially, the two-dimensional weighted sieve gives $N=P_2+P_3$ for every large even $N$ (Wang Yuan, 1957).
\end{remark}

\section{The uniform bound and the Poincar\'e series}\label{sec:spectral}

Let $L=4mm'$ and let $P_N\in S_2(\Gamma_0(L))$, $N\ge1$, be the Riesz representative of the $N$-th coefficient: the unique cusp form with
\begin{equation}\label{eq:riesz}
\langle f,P_N\rangle=\frac{a_f(N)}{4\pi N}\qquad\text{for all }f\in S_2(\Gamma_0(L)),
\end{equation}
i.e.\ $P_N=(4\pi N)^{-1}\sum_{f\in\mathcal B}\overline{a_f(N)}\,f$ for an orthonormal basis $\mathcal B$. (This is the $N$-th Poincar\'e series; for weight $2$ the series itself converges only conditionally, which is why we define $P_N$ by \eqref{eq:riesz}.) By (11) of Part~I,
\[
\|P_N\|^2=\frac{a_{P_N}(N)}{4\pi N}=\frac{1}{(4\pi N)^2}\sum_{f\in\mathcal B}|a_f(N)|^2=\frac{\Delta_L(N,N)}{4\pi N},
\]
where $\Delta_L(N,N)$ is the Petersson density. Hence, for $G=G_{m,m'}$,
\begin{equation}\label{eq:angle}
|a_G(N)|^2=(4\pi N)^2|\langle G,P_N\rangle|^2=4\pi N\,\Delta_L(N,N)\,\|G\|^2\cos^2\angle(G,P_N),
\end{equation}
where $\cos\angle(G,P_N)=|\langle G,P_N\rangle|/(\|G\|\,\|P_N\|)$. The identity separates the three quantities that enter any bound for $a_G(N)$: the norm $\|G\|^2$, which by Theorem~\ref{thm:norm}(i) is $\ll L^\e$; the Petersson density, which is $\le C'$ for $N\le L^{1/2}$ (Part~I, Lemma~2.2, Step~4) and $1+O(N^{1/2+\e}/L)$ in general; and the alignment $\cos^2\angle(G,P_N)\le1$. Theorem~\ref{thm:norm}(ii) is obtained from \eqref{eq:angle} by bounding the alignment trivially: after the coprime part of $N$ is moved into a Hecke operator (Part~I, Lemma~2.2), only $P_{n_2}$ with $n_2\in\{1,2\}$ occurs, where the density is bounded, and the norm does the rest. Theorem~\ref{thm:A} says what the alignment is on average: inserting \eqref{eq:angle} into \eqref{eq:A} and restricting to $N\in(Y/2,Y]$,
\begin{equation}\label{eq:avgangle}
\frac{1}{Y/2}\sum_{Y/2<N\le Y}\Delta_L(N,N)\cos^2\angle(G,P_N)\ \le\ 4e^2\Bigl(\frac1Y+\frac{54}L\Bigr)\ \ll\ \frac1L\qquad(Y\ge L),
\end{equation}
the value of $\cos^2$ for a unit vector in general position in the $\asymp L$-dimensional space $S_2(\Gamma_0(L))$: on average over $N$, the theta series is no more aligned with the Poincar\'e series than a random cusp form of level $L$ would be, and Theorem~\ref{thm:norm}(ii) says that no single $N$ is aligned by more than $N^\e L/\|G\|^2\gg L^{1-\e}$ times the average. Expanding in an orthogonal basis $\mathcal B^*$ of Hecke eigenforms as in Part~I, Section~5.5,
\[
\langle G,P_N\rangle=\sum_{\varphi\in\mathcal B^*}\frac{\langle G,\varphi\rangle\langle\varphi,P_N\rangle}{\|\varphi\|^2}=\frac{1}{4\pi N}\sum_{\varphi\in\mathcal B^*}\frac{\langle\Theta,\varphi\rangle\,a_\varphi(N)}{\|\varphi\|^2},
\]
so that Theorem~\ref{thm:norm}(ii) is the bound $\ll N^{1/2+\e}$ for this spectral sum, uniformly in $m,m'$; the first two versions of Part~I asked for a proof through the cancellation between the coefficients $\langle\Theta,\varphi\rangle$ and $a_\varphi(N)$. The proof in Part~I does not see this cancellation: it shows instead that $\sum_\varphi|\langle\Theta,\varphi\rangle|^2/\|\varphi\|^2=\|G\|^2$ is itself small, i.e.\ that the theta series has a small cuspidal projection to begin with, and the spectral sum is then bounded by Cauchy--Schwarz.

\subsection{The individual bound}\label{subsec:ind}
Theorem~\ref{thm:A} and the data (Section~\ref{subsec:sup}) say that the cancellation which the proof does not see is there for most $N$, and the sieve for every $N$ would profit from it for each $N$. We state the strongest form suggested by the data.

\begin{conjecture}\label{conj:ind}
For coprime odd $m,m'$ and $N\ge1$ with $(N,mm')=1$ and $v_2(N)\le1$,
\[
|T(m,m';N)-M(m,m';N)|=|a_{G_{m,m'}}(N)|\ \ll_\e\ N^{1/2+\e}(mm')^{-1/2+\e}.
\]
In the terms of \eqref{eq:angle}: $\Delta_L(N,N)\cos^2\angle(G,P_N)\ll N^\e L^{-1+\e}/\|G\|^2$, i.e.\ no Poincar\'e series $P_N$ is aligned with $G$ by more than $N^\e L^\e$ times the average \eqref{eq:avgangle}.
\end{conjecture}

Conjecture~\ref{conj:ind} is the mean-square bound \eqref{eq:A2} applied to each $N$ separately; it is consistent with the trivial lower bound $|a_G(N)|\ge\mathrm{dist}(M,8\Z)$ for $N\ll L$, where $M\asymp N/(mm')\ll1$, and with the data of Section~\ref{subsec:sup}, where the supremum over $10^6$ to $4\cdot10^6$ values of $N$ exceeds the root mean square by a factor between $5$ and $10$ at every level computed. We do not know how to prove any power saving in $mm'$ for an individual $N$. In the spectral form the question is a twisted first moment $\sum_\varphi\langle\Theta,\varphi\rangle a_\varphi(N)/\|\varphi\|^2$ over the forms of level $L$ with the ``long'' twist $N\ge L$ (the sieve at the level $\theta$ uses $N\approx L^{1/\theta}$), outside the range $N\le L^{1-\delta}$ in which the Petersson formula evaluates such moments. Any saving would raise the level for every $N$:

\begin{proposition}\label{prop:beyond}
Let $0\le\eta\le\frac12$ and suppose that $|a_{G_{m,m'}}(N)|\ll_\e N^{1/2+\e}(mm')^{-\eta+\e}$ for all coprime odd $m,m'$ and all $N$ with $(N,mm')=1$, $v_2(N)\le1$. Then Theorem~3.2 of Part~I holds for every $N\equiv2\pmod 4$ with $D=N^{\theta-\e}$, $\theta=1/(2(1-\eta))$, i.e.\ $r(n)r(N-n)$ has level of distribution $1/(2(1-\eta))$. In particular, Conjecture~\ref{conj:ind} gives the level $1-\e$ for every $N$, and then the sieve of Section~\ref{sec:B} gives, for every sufficiently large $N\equiv2\pmod4$, $\Omega(n(N-n))\le5$ and $\max(\Omega(n),\Omega(N-n))\le4$ for at least $0.34\,\Sg_r(N)N/(\log N)^2$ integers $n$.
\end{proposition}
\begin{proof}
We follow the proof of Theorem~3.2 of Part~I and only change the bound for one remainder. Each term of $R(d,e)$ is $\pm c\,a_{G}(N')$ for a form $G=G_{\tilde m,\tilde m'}$ with $\tilde m\tilde m'=t_1ss'\cdot de$, where $t\mid N^\flat$, $t_1\mid t^2$, and $N'\in\{N,N/2\}$ may have common factors with $\tilde m\tilde m'$ at the primes of $t$ only. Lemma~2.5 of Part~I reduces $a_G(N')$ to $\le4^{\omega(N)}d(N)$ terms $a_{G_{m_j,m_j'}}(N_j)$ with $(N_j,m_jm_j')=1$, $N_j\mid N'$ and $m_jm_j'\ge\tilde m\tilde m'/t_1^{2}\ge de/t_1^2$ (each reduction step at a prime $p$ divides the product of the coefficients by at most $p^2$, and only the primes of $t_1$ are involved). Hence, by the hypothesis with $\e/20$ in place of $\e$,
\[
|a_G(N')|\ll 4^{\omega(N)}d(N)\,N^{1/2+\e/20}\,(de)^{-\eta}\,t_1^{2\eta}\,(mm')^{\e/20}.
\]
Put $\eta'=\e/20$. For the terms with $t_1ss'\le N^\e$ this is $\ll N^{1/2+3\eta'+2\e\eta}(de)^{-\eta}$ by (F1) of Part~I, and their total is
\[
\ll N^{1/2+5\eta'+\e}\sum_{k\le D}d(k)k^{-\eta}\ll N^{1/2+6\eta'+\e}D^{1-\eta}=N^{1/2+6\eta'+\e}N^{1/2-\e(1-\eta)}\le N^{1-\e/2+6\eta'},
\]
since $D^{1-\eta}=N^{(\theta-\e)(1-\eta)}=N^{1/2-\e(1-\eta)}$ and $1-\eta\ge\frac12$. The terms with $t_1ss'>N^\e$ are bounded by the trivial bound \eqref{eq:triv} exactly as in Part~I, by $N^{1-\e+5\eta'}+N^{\theta+5\eta'}$. With $\theta\le1-\e$ (replace $\e$ by $\min(\e,1-\theta)$ if necessary) the total is $\ll N^{1-\e/2+6\eta'}\le N^{1-\e/5}$, which is the conclusion of Theorem~3.2 of Part~I with $N^{1-\e/5}$ in place of $N^{1-\e/3}$; Proposition~4.3 of Part~I needs only a power saving. The last assertion is Section~\ref{sec:B}, where the parameters at the level $19/20$ were certified for any $N$ satisfying \eqref{eq:B}.
\end{proof}

The exponent $\theta=1/(2(1-\eta))$ interpolates between the level $1/2$ of Part~I ($\eta=0$) and the level $1$ ($\eta=\frac12$); the level $2/3$, which gives $(6,4)$ for every $N$ with the certified parameters of Section~\ref{sec:B}, needs only $\eta=\frac14$, i.e.\ a saving of $(mm')^{-1/4}$ over Theorem~\ref{thm:norm}(ii), half of what the mean square \eqref{eq:A2} gives on average.

\subsection{An exact check at the level $20$}\label{subsec:exact}
The space $S_2(\Gamma_0(20))$ is one-dimensional, spanned by the newform $f_{20}=\eta(2z)^2\eta(10z)^2$ of the elliptic curve $20a1$: $y^2=x^3+x^2+4x+4$, which is $X_0(20)$ itself (modular degree $1$, Manin constant $1$). Hence $G_{1,5}=c\,f_{20}$ with $c=a_G(1)=T(1,5;1)-M(1,5;1)=4-\tfrac43=\tfrac83$ (the Siegel product at $n=1$ is $\frac{\pi^2}5\cdot\frac{25}{3\pi^2}\cdot\frac45$), and $\|G_{1,5}\|^2=\frac{64}9\|f_{20}\|^2$. The script \texttt{norm\_level20.py} computes $\|f_{20}\|^2$ in two independent ways: from the Rankin--Selberg asymptotic $\sum_{n\le y}a_f(n)^2/n\sim16\pi^2\|f\|^2y/\vol$, $\vol(\Gamma_0(20)\backslash\HH)=12\pi$, with the coefficients of the eta product up to $2^{20}$, which gives $0.08128$, $0.08128$, $0.08134$ at $y=2^{16},2^{18},2^{20}$; and from the period lattice $\Lambda$ of $20a1$, since $\varphi^*\omega=2\pi if(z)\dd z$ gives $\|f\|^2=\mathrm{covol}(\Lambda)/(4\pi^2)$ for a modular parametrisation of degree $1$. The periods $\omega_1=2.824375\ldots$ (real) and $\omega_2=2.274165\ldots i$ of $y^2=(x+1)(x^2+4)$ are computed by quadrature, the lattice $\Z\omega_1+\Z(\omega_1+\omega_2)/2$ is identified as the period lattice by recomputing $g_2=-44/3$ and $|g_3|=296/27$ from it through $E_4$ and $E_6$, and its covolume $3.2115478\ldots$ gives $\|f_{20}\|^2=0.0813495\ldots$. The two values agree to four decimals. The script also checks that $T(1,5;N)-M(1,5;N)$ as computed by the pipeline of Part~I equals $\frac83a_f(N)$ on every admissible $N\le2^{20}$, to $10^{-9}$. This validates the Siegel main term, the FFT computation, the Rankin--Selberg normalisation and the volume used in the estimate $\|G\|^2_{\mathrm{est}}$. The exact value $\|G_{1,5}\|^2=0.5785$ is below the estimate $0.74$ of Table~\ref{tab:ms}, as expected: the excluded $N$ are either multiples of $5$, where $|a_f(5n)|=|a_f(n)|$, or divisible by $16$, where $a_f=0$, so the admissible mean is biased upwards.

\subsection{The empirical law}\label{subsec:law}
The last column of Table~\ref{tab:ms} shows $\|G\|^2_{\mathrm{est}}/(\log mm')^2$. For the five pairs with $10^2\le mm'\le1.1\cdot10^4$, $Y\ge50L$ and $mm'$ free of small prime factors, it lies between $0.348$ and $0.375$, with mean $0.365$, while the ratio to $(mm')^{1/2}$ falls from $0.79$ to $0.29$ and the ratio to $(mm')^{1/4}$ rises from $2.5$ to $3.2$ over the same range. Over a range of a factor $100$ in $mm'$ the fits $(\log mm')^2$ and $(mm')^{1/4}$ are not separated by the data alone, but Theorem~\ref{thm:norm}(i) excludes any power, and the heuristic of Part~I, Remark~2.13, that $\|G\|^2$ is a sum over the cusps of $\sum_{n\ll mm'}|b(n)|^2/n$ with $|b(n)|\approx r(\alpha)r(\beta)$ on $O(1)$ pairs, predicts $(\log mm')^2$ with a constant of the order of $\frac1{4\pi}\cdot2\cdot(\text{number of cusp classes with a large contribution})\approx0.3$--$0.5$. We therefore record
\[
\|G_{m,m'}\|^2\approx0.37\,(\log mm')^2\qquad(10^2\le mm'\le1.6\cdot10^4),
\]
and, for the root mean square of $(T-M)/\sqrt N$ over $N\le Y$, $Y\gg L$,
\[
\mathrm{rms}^2\approx\frac{16\pi^2\|G\|^2}{\vol(\Gamma_0(L)\backslash\HH)}\approx\frac{44\,(\log mm')^2}{mm'\prod_{p\mid 2mm'}(1+1/p)},
\]
which reproduces the column ``rms'' of Table~3 of Part~I ($0.27$ for $(101,103)$, $0.67$ for $(1,1009)$). This corrects the remark in Section~5.3 of the first two versions of Part~I that the root mean square ``decreases like $(mm')^{-1/4}$'' and that Waibel's bound ``appears to be sharp'': the decrease is like $(mm')^{-1/2}$ up to a logarithm, as Theorem~\ref{thm:A} with Theorem~\ref{thm:norm}(i) shows; version~3 of Part~I states the corrected law.

\subsection{The supremum over $N$}\label{subsec:sup}
Conjecture~\ref{conj:ind} says that the supremum of $|a_G(N)|/\sqrt N$ over $N$ in a range is, up to $N^\e L^\e$, of the size of the root mean square $\approx2.7(\log mm')(mm')^{-1/2}$ of Section~\ref{subsec:law}. Table~\ref{tab:sup}, extracted from the scans of Part~I (\texttt{results/sup\_error\_scan\_x21.txt}, \texttt{\_x23.txt} there), gives the ratio of the supremum over all admissible $N\in(X/2,X]$ to the root mean square, for $X=2^{21}$ (about $10^6$ values of $N$ per pair) and $X=2^{23}$ (about $4\cdot10^6$). For the pairs with $L\le6\cdot10^4$ the ratio lies between $5.4$ and $9.8$ and shows no trend in $mm'$; its mild growth from $X=2^{21}$ to $2^{23}$ is the growth of the extreme value of a fixed heavy-tailed distribution with the sample size (the suprema are attained at $N$ with $16$--$192$ divisors, in line with the factor $d(N)$ of Deligne's bound). The pair $(9,1)$ is special: $S_2(\Gamma_0(36))$ is spanned by the newform of the curve $36a1$, $y^2=x^3+1$, which has complex multiplication, so that $a_f(p)=0$ for $p\equiv2\pmod3$ and the distribution of $a_f(N)$ is far from the Gaussian bulk of Figure~2 of Part~I. The pair $(1009,1013)$ is in the range $X<2L$, where neither the mean square nor the supremum has reached its asymptotic regime.
